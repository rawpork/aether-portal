export interface Env {
  DB: D1Database;
}

export default {
  async fetch(request: Request, env: Env): Promise<Response> {
    const url = new URL(request.url);
    const corsHeaders = {
      "Access-Control-Allow-Origin": "*",
      "Access-Control-Allow-Methods": "GET, POST, PATCH, OPTIONS",
      "Access-Control-Allow-Headers": "Content-Type, Authorization",
    };

    if (request.method === "OPTIONS") {
      return new Response(null, { headers: corsHeaders });
    }

    // 1. Authentication: Google OAuth Callback
    if (url.pathname === "/api/auth/google" && request.method === "POST") {
      try {
        const { credential } = await request.json() as any;
        if (!credential) {
          return new Response(JSON.stringify({ error: "Missing credential token" }), { status: 400, headers: corsHeaders });
        }
        return new Response(JSON.stringify({ success: true, message: "Authenticated successfully" }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      } catch (err: any) {
        return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: corsHeaders });
      }
    }

    // 2. Authentication: Logout
    if (url.pathname === "/api/auth/logout" && request.method === "POST") {
      return new Response(JSON.stringify({ success: true, message: "Logged out" }), {
        headers: { ...corsHeaders, "Content-Type": "application/json" }
      });
    }

    // 3. User Settings: Update Preferred Name
    if (url.pathname === "/api/settings" && request.method === "PATCH") {
      try {
        const { preferred_name } = await request.json() as any;
        return new Response(JSON.stringify({ success: true, preferred_name: preferred_name || null }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      } catch (err: any) {
        return new Response(JSON.stringify({ error: err.message }), { status: 500, headers: corsHeaders });
      }
    }

    // 4. Tool Registry: Search Endpoint (Case-Insensitive SQL Matching)
    if (url.pathname === "/api/tools/search") {
      const query = (url.searchParams.get("q") || "").toLowerCase().trim();
      const limit = parseInt(url.searchParams.get("limit") || "20");

      let sql = "SELECT tool_id, tool_type, name, description, args_json, file_path FROM tools_manifest";
      let params: any[] = [];

      if (query) {
        sql += " WHERE LOWER(name) LIKE ? OR LOWER(description) LIKE ? OR LOWER(tool_id) LIKE ?";
        const searchTerm = `%${query}%`;
        params = [searchTerm, searchTerm, searchTerm];
      }

      sql += " ORDER BY name ASC LIMIT ?";
      params.push(limit);

      try {
        const stmt = env.DB.prepare(sql).bind(...params);
        const { results } = await stmt.all();
        return new Response(JSON.stringify({ success: true, count: results.length, tools: results }), {
          headers: { ...corsHeaders, "Content-Type": "application/json" }
        });
      } catch (err: any) {
        return new Response(JSON.stringify({ success: false, error: err.message }), { status: 500, headers: corsHeaders });
      }
    }

    // 5. Tool Registry: Execute Proxy
    if (url.pathname === "/api/tools/execute" && request.method === "POST") {
      try {
        const { tool_id, payload } = await request.json() as any;
        const stmt = env.DB.prepare("SELECT * FROM tools_manifest WHERE tool_id = ?").bind(tool_id);
        const tool: any = await stmt.first();

        if (!tool) {
          return new Response(JSON.stringify({ success: false, error: `Tool '${tool_id}' not found` }), { status: 404, headers: corsHeaders });
        }

        return new Response(JSON.stringify({
          success: true,
          status: "DISPATCHED",
          tool_id: tool.tool_id,
          tool_type: tool.tool_type,
          file_path: tool.file_path,
          payload: payload || {}
        }), { headers: { ...corsHeaders, "Content-Type": "application/json" } });
      } catch (err: any) {
        return new Response(JSON.stringify({ success: false, error: err.message }), { status: 500, headers: corsHeaders });
      }
    }

    return new Response("Aether Engine Core API Gateways Operational", { status: 200, headers: corsHeaders });
  }
};
